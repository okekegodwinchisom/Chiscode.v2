_AGENTS_MD_LOAD_TIME: float = 0.0


def load_agents_md() -> str:
    """
    Load the AGENTS.md file from the project root.
    This serves as the authoritative guide for all code generation.
    """
    global _AGENTS_MD_CACHE, _AGENTS_MD_VERSION, _AGENTS_MD_LOAD_TIME
    
    if _AGENTS_MD_CACHE is None:
        # Look for AGENTS.md in multiple locations
        search_paths = [
            Path(__file__).parent.parent.parent / "AGENTS.md",  # Project root
            Path(__file__).parent.parent / "AGENTS.md",        # Agents directory
            Path.cwd() / "AGENTS.md",                          # Current working directory
        ]
        
        for path in search_paths:
            if path.exists():
                _AGENTS_MD_CACHE = path.read_text(encoding="utf-8")
                _AGENTS_MD_VERSION = _extract_agents_version(_AGENTS_MD_CACHE)
                _AGENTS_MD_LOAD_TIME = path.stat().st_mtime
                break
        
        # Fallback: embedded AGENTS.md content
        if _AGENTS_MD_CACHE is None:
            _AGENTS_MD_CACHE = _get_fallback_agents_md()
            _AGENTS_MD_VERSION = "v0.0.0-fallback"
    
    return _AGENTS_MD_CACHE


def _extract_agents_version(content: str) -> str:
    """Extract version from AGENTS.md header."""
    for line in content.splitlines():
        if "Version:" in line:
            return line.split("Version:")[-1].strip()
        if "v" in line and "." in line and len(line) < 30:
            # Try to find version pattern like v1.0.0
            import re
            match = re.search(r'v\d+\.\d+\.\d+', line)
            if match:
                return match.group()
    return "v1.0.0"

def _load_generation_skill() -> str:
    """
    Load the SkillOpt-optimized generation skill from the skill file.
    Falls back to base skill if optimized version doesn't exist.
    """
    global _SKILL_CACHE, _SKILL_VERSION, _SKILL_LOAD_TIME
    
    # Refresh if SKILL_REFRESH env var is set
    if os.getenv("SKILL_REFRESH"):
        _SKILL_CACHE = None
    
    if _SKILL_CACHE is None:
        skill_dir = Path(__file__).parent.parent / "skills"
        optimized = skill_dir / "chiscode_skill_optimized.md"
        base = skill_dir / "chiscode_skill.md"
        
        if optimized.exists():
            _SKILL_CACHE = optimized.read_text()
            _SKILL_VERSION = _extract_version(_SKILL_CACHE)
            _SKILL_LOAD_TIME = optimized.stat().st_mtime
        elif base.exists():
            _SKILL_CACHE = base.read_text()
            _SKILL_VERSION = "v1.0.0"
            _SKILL_LOAD_TIME = base.stat().st_mtime
        else:
            _SKILL_CACHE = _get_fallback_skill()
            _SKILL_VERSION = "v0.0.0-fallback"
    
    return _SKILL_CACHE

def _extract_version(skill_content: str) -> str:
    """Extract version from skill file header."""
    for line in skill_content.splitlines():
        if "Version:" in line:
            return line.split("Version:")[-1].strip()
    return "v1.0.0"

def _get_stack_specific_rules(stack: dict, auth_required: bool, database: str) -> str:
    """
    Get stack-specific rules from stack_rules.py.
    This is the DETAILED rules that were previously not being included.
    """
    fe = _get_stack_string(stack, "frontend")
    be = _get_stack_string(stack, "backend")
    
    rules = []
    
    # FastAPI rules
    if _is_fastapi_stack(stack) or _is_python_stack(stack):
        has_alembic = "alembic" in " ".join(_normalize_extras(stack.get("extras", []))).lower()
        has_websockets = any(x in " ".join(_normalize_extras(stack.get("extras", []))).lower() 
                            for x in ("websocket", "realtime", "socket"))
        has_durable = "pg_durable" in " ".join(_normalize_extras(stack.get("extras", []))).lower()
        
        rules.append(get_fastapi_rules(
            has_alembic=has_alembic,
            has_websockets=has_websockets,
            has_auth=auth_required,
            has_durable=has_durable,
        ))
    
    # HTMX rules
    if _is_htmx_stack(stack):
        rules.append(get_htmx_rules())
    
    # Alpine rules
    if _is_alpine_stack(stack):
        rules.append(get_alpine_rules())
    
    # Next.js rules
    if _is_nextjs_stack(stack):
        rules.append(get_nextjs_rules())
    
    # React rules
    if _is_react_stack(stack):
        rules.append(get_react_rules())
    
    # SvelteKit rules
    if _is_sveltekit_stack(stack):
        rules.append(get_sveltekit_rules())
    
    # Database rules
    if database and database.lower() not in ("none", ""):
        rules.append(get_db_rules(database, stack))
    
    return "\n\n".join(r for r in rules if r)


def _extract_anti_patterns_from_skill() -> str:
    """Extract anti-patterns from the skill file."""
    skill = _load_generation_skill()
    lines = skill.splitlines()
    in_anti = False
    anti_lines = []
    
    for line in lines:
        if "ANTI-PATTERNS" in line:
            in_anti = True
            continue
        if in_anti:
            if line.startswith("## ") and "ANTI" not in line:
                break
            anti_lines.append(line)
    
    return "\n".join(anti_lines) if anti_lines else ""

def _normalize_extras(extras_raw: Any) -> List[str]:
    """Normalize extras to a list of strings."""
    if not extras_raw:
        return []
    if isinstance(extras_raw, str):
        return [extras_raw]
    if isinstance(extras_raw, (list, tuple)):
        return [str(item) for item in extras_raw]
    return []


def _normalize_features(features_raw: Any) -> List[str]:
    """Normalize features to a list of strings."""
    if not features_raw:
        return []
    if isinstance(features_raw, str):
        return [features_raw]
    if isinstance(features_raw, (list, tuple)):
        return [str(item) for item in features_raw]
    return []


def _get_stack_string(stack: dict, key: str) -> str:
    """Safely get a string value from stack dict."""
    value = stack.get(key, "")
    return str(value).lower() if value else ""


def _is_framework_in_stack(stack: dict, keywords: tuple) -> bool:
    """Check if any framework keywords are present in the stack."""
    frontend = _get_stack_string(stack, "frontend")
    backend = _get_stack_string(stack, "backend")
    extras = " ".join(_normalize_extras(stack.get("extras", []))).lower()
    combined = f"{frontend} {backend} {extras}"
    return any(keyword in combined for keyword in keywords)


def _is_nextjs_stack(stack: dict) -> bool:
    return _is_framework_in_stack(stack, NEXTJS_KEYWORDS)


def _is_sveltekit_stack(stack: dict) -> bool:
    return _is_framework_in_stack(stack, SVELTEKIT_KEYWORDS)


def _is_react_stack(stack: dict) -> bool:
    return _is_framework_in_stack(stack, REACT_KEYWORDS) and not _is_nextjs_stack(stack)


def _is_fastapi_stack(stack: dict) -> bool:
    return _is_framework_in_stack(stack, FASTAPI_KEYWORDS)


def _is_python_stack(stack: dict) -> bool:
    return _is_framework_in_stack(stack, PYTHON_KEYWORDS)


def _is_node_stack(stack: dict) -> bool:
    return _is_framework_in_stack(stack, NODE_KEYWORDS)


def _is_rust_stack(stack: dict) -> bool:
    return _is_framework_in_stack(stack, RUST_KEYWORDS)


def _is_htmx_stack(stack: dict) -> bool:
    return _is_framework_in_stack(stack, HTMX_KEYWORDS)


def _is_alpine_stack(stack: dict) -> bool:
    return _is_framework_in_stack(stack, ALPINE_KEYWORDS)

# Complexity Classifier
# ─────────────────────────────────────────────────────────────────────────────

def _classify_complexity(
    features: List[str],
    auth_required: bool,
    database: str,
    stack: dict,
) -> Complexity:
    """Classify project complexity based on features, auth, and stack extras."""
    score = 0
    features_list = _normalize_features(features)
    extras = _normalize_extras(stack.get("extras", []))
    extras_str = " ".join(extras).lower()
    
    # Base score from number of features
    score += len(features_list)
    
    # Auth adds significant complexity
    if auth_required:
        score += 3
    
    # Database adds complexity
    if database and database.lower() not in ("none", ""):
        score += 2
    
    # ORMs add complexity
    if any(kw in extras_str for kw in PRISMA_KEYWORDS + DRIZZLE_KEYWORDS + SQLALCHEMY_KEYWORDS):
        score += 2
    
    # Payment integrations add significant complexity
    if any(kw in extras_str for kw in PAYMENT_KEYWORDS):
        score += 3
    
    # Queues/workers add complexity
    if any(kw in extras_str for kw in QUEUE_KEYWORDS):
        score += 2
    
    # WebSockets/realtime add complexity
    if any(kw in extras_str for kw in REALTIME_KEYWORDS):
        score += 2
    
    # File storage adds slight complexity
    if any(kw in extras_str for kw in STORAGE_KEYWORDS):
        score += 1
    
    # Multiple frameworks add complexity
    framework_count = sum([
        _is_nextjs_stack(stack),
        _is_fastapi_stack(stack),
        _is_sveltekit_stack(stack),
        _is_rust_stack(stack),
    ])
    if framework_count >= 2:
        score += 2
    
    if score <= 4:
        return Complexity.SIMPLE
    elif score <= 9:
        return Complexity.MODERATE
    else:
        return Complexity.COMPLEX

def _get_immutable_stack_block(stack: dict, auth_required: bool) -> str:
    """
    Locked stack rules block — only correctness guarantees.
    These are NEVER optimized by SkillOpt.
    """
    fe = _get_stack_string(stack, "frontend")
    be = _get_stack_string(stack, "backend")
    
    sections = [
        "# ═══════════════════════════════════════════════════════════",
        "# STACK CONSTRAINTS — LOCKED (IMMUTABLE)",
        "# ═══════════════════════════════════════════════════════════",
    ]
    
    # CDN-only rule (critical — prevents package.json generation for HTMX projects)
    if "htmx" in fe or "alpine" in fe:
        if not any(x in fe for x in ("next", "react", "vue", "svelte", "nuxt")):
            sections.append(
                "CDN-ONLY PROJECT: NO package.json, NO npm, NO node_modules. "
                "Load all JS via <script> tags."
            )
    
    # Framework-specific critical rules — keep these under 200 chars each
    if "fastapi" in be or "python" in be:
        sections.append(
            "FASTAPI: sync get_db() → Generator[Session], sync init_db(), "
            "DeclarativeBase NOT declarative_base(), model_dump() NOT .dict(), "
            "extra='ignore' NOT 'forbid'"
        )
    
    if "svelte" in fe:
        sections.append(
            "SVELTEKIT: NEVER access localStorage/window/document at module level. "
            "Use import { browser } from '$app/environment'; if (browser) { ... }"
        )
    
    if "next" in fe:
        sections.append(
            "NEXT.JS: Use App Router (app/), NOT Pages Router. "
            "globals.css ONLY in app/globals.css. Server Components by default."
        )
    
    return "\n\n".join(sections)

def _get_auth_rules(stack: dict, complexity: Complexity) -> str:
    """Generate authentication rules based on the stack and complexity level."""
    extras = _normalize_extras(stack.get("extras", []))
    extras_str = " ".join(extras).lower()
    
    is_nextjs = _is_nextjs_stack(stack)
    is_fastapi = _is_fastapi_stack(stack)
    is_nextauth = any(kw in extras_str for kw in NEXTAUTH_KEYWORDS)
    is_jwt = any(kw in extras_str for kw in JWT_KEYWORDS)
    is_prisma = any(kw in extras_str for kw in PRISMA_KEYWORDS)
    
    lines = ["=== AUTH REQUIREMENTS ==="]
    
    if is_nextjs:
        if is_nextauth:
            lines += [
                "- Use next-auth v5 (Auth.js). Config in app/api/auth/[...nextauth]/route.ts",
                "- Export { handlers: { GET, POST } } from auth config",
                "- Use getServerSession(authOptions) in server components and API routes",
                "- Create middleware.ts to protect routes — use auth() wrapper",
                "- Store session data in JWT strategy unless database sessions specified",
                "- NEVER expose session secrets client-side",
                "- Provide AuthProvider wrapper component for client components",
            ]
            if is_prisma:
                lines += [
                    "- Use PrismaAdapter for database session storage",
                    "- Add Session and Account models to schema.prisma",
                ]
        elif is_jwt:
            lines += [
                "- Implement JWT auth with jose library",
                "- Store access token in httpOnly cookie (NOT localStorage)",
                "- Implement refresh token rotation",
                "- Create middleware.ts for route protection",
                "- API routes must validate JWT before processing",
            ]
        else:
            lines += [
                "- Implement session-based auth with httpOnly cookies",
                "- Create /api/auth/login, /api/auth/logout, /api/auth/me endpoints",
                "- Protect routes via middleware.ts using NextRequest/NextResponse",
                "- Use crypto.randomBytes for session token generation",
            ]
    
    elif is_fastapi:
        lines += [
            "- Use python-jose for JWT, passlib[bcrypt] for password hashing",
            "- Create app/core/auth.py with get_current_user dependency",
            "- OAuth2PasswordBearer scheme for token extraction",
            "- /auth/login returns access_token + refresh_token",
            "- /auth/refresh accepts refresh_token cookie, returns new access_token",
            "- Protect routes with Depends(get_current_user)",
            "- Store refresh tokens in database for revocation support",
        ]
    
    else:
        lines += [
            "- Implement secure authentication using best practices for the chosen stack",
            "- Store session data securely (httpOnly cookies, server-side sessions)",
            "- Protect sensitive endpoints with authentication middleware",
            "- Hash passwords using industry-standard algorithms (bcrypt, Argon2)",
            "- Implement proper logout functionality that invalidates sessions",
        ]
    
    if complexity == Complexity.COMPLEX:
        lines += [
            "- Implement role-based access control (RBAC) if multiple user types",
            "- Log auth events (login, logout, failed attempts) to database",
            "- Rate limit auth endpoints: max 5 attempts per minute per IP",
            "- Implement account lockout after 5 failed attempts",
        ]
    
    return "\n".join(lines)

# API Route + Database Init Rules
# ─────────────────────────────────────────────────────────────────────────────

def _get_api_and_db_init_rules(stack: dict, database: str) -> str:
    """
    Generate API route and database initialization rules.
    
    Crucial for preventing common API response shape mismatches and
    database initialization failures.
    """
    is_nextjs = _is_nextjs_stack(stack)
    db = (database or "").lower()
    
    is_turso = any(x in db for x in ("turso", "sqlite", "libsql"))
    is_pg = any(x in db for x in ("postgres", "postgresql", "neon", "supabase"))
    
    if not is_nextjs and not is_turso and not is_pg:
        return ""
    
    lines = ["=== API ROUTE + DATABASE INIT RULES ==="]
    
    if is_nextjs:
        lines += [
            "",
            "── Next.js API Routes ──",
            "POST endpoints MUST return the complete created object, not just {id}:",
            "  ✓ USE:  return NextResponse.json(result.rows[0], { status: 201 })",
            "  ✗ NEVER: return NextResponse.json({ id: result.lastInsertRowid })",
            "  Reason: page.tsx does setItems([...items, data]) — data must match Item type",
            "",
            "Dynamic route handlers MUST include typed params:",
            "  ✓ USE:  async function PUT(req: Request, { params }: { params: { id: string } })",
            "  ✗ NEVER: async function PUT(req: Request) { const { id } = await req.json() }",
            "  Reason: dynamic segment [id] comes from params, NOT the request body",
            "",
            "GET endpoints returning lists MUST return plain arrays:",
            "  ✓ USE:  return NextResponse.json(result.rows)",
            "  ✗ NEVER: return NextResponse.json({ data: result.rows })",
            "  Reason: client does setItems(data) — wrapped object = nothing renders",
            "",
            "Client-side data fetching must guard against non-array responses:",
            "  setItems(Array.isArray(data) ? data : [])",
        ]
    
    if is_turso or is_pg:
        lines += [
            "",
            "── Database Initialization ──",
            "ALWAYS create tables using CREATE TABLE IF NOT EXISTS in lib/init-db.ts.",
            "NEVER assume the database has tables — the app must self-initialize.",
            "Call initDb() at the top of the first API route or in a startup hook.",
        ]
    
    if is_turso:
        lines += [
            "",
            "── Turso/SQLite specifics ──",
            "completed fields: store as INTEGER (0/1), normalize to boolean in GET responses:",
            "  completed: row.completed === 1 || row.completed === true",
            "Use RETURNING * after INSERT/UPDATE to get the full row back.",
            "Use parameterized args: [] — never string interpolation in SQL.",
            "Env vars: DATABASE_URL=libsql://..., DATABASE_AUTH_TOKEN=...",
        ]
    
    if is_pg:
        lines += [
            "",
            "── PostgreSQL specifics ──",
            "Use SERIAL PRIMARY KEY or uuid_generate_v4() for IDs.",
            "Use RETURNING * after INSERT to get the full created row.",
            "Initialize schema in lib/init-db.ts with CREATE TABLE IF NOT EXISTS.",
            "Use connection pooling for serverless environments.",
        ]
    
    return "\n".join(lines)

# Architecture Rules
# ─────────────────────────────────────────────────────────────────────────────

def _get_architecture_rules(
    features: List[str],
    stack: dict,
    complexity: Complexity,
) -> str:
    """
    Generate architecture guidance based on stack and complexity.
    
    Returns empty string for simple projects.
    """
    if complexity == Complexity.SIMPLE:
        return ""
    
    is_nextjs = _is_nextjs_stack(stack)
    is_fastapi = _is_fastapi_stack(stack)
    is_sveltekit = _is_sveltekit_stack(stack)
    is_react = _is_react_stack(stack)
    
    lines = ["=== ARCHITECTURE REQUIREMENTS ==="]
    
    if is_nextjs:
        lines += [
            "- Separate concerns: components/ for UI, lib/ for business logic, app/api/ for endpoints",
            "- Server Components by default — add 'use client' only when needed",
            "- Data fetching in Server Components using async/await directly",
            "- Use React Server Actions for form submissions when possible",
            "- API routes for external consumption or when client-side fetch is required",
            "- lib/actions.ts: Server Actions with 'use server' directive",
        ]
    
    elif is_fastapi:
        lines += [
            "- Follow feature-based folder structure: src/routers/, src/models/, src/schemas/, src/services/",
            "- Pydantic schemas in schemas/: separate Request and Response schemas",
            "- Business logic in services/ — not in route handlers",
            "- Database operations via dependency injection: Depends(get_db)",
            "- Use APIRouter with prefix and tags for each resource",
            "- Ensure imports and schemas correspondence between files",
        ]
    
    elif is_sveltekit:
        lines += [
            "- Follow SvelteKit best practices: +page.svelte, +layout.svelte, +page.server.ts",
            "- Use load functions for data fetching in +page.ts or +page.server.ts",
            "- Keep components small and focused",
            "- Use stores for shared state when needed",
        ]
    
    elif is_react:
        lines += [
            "- Use feature-based folder structure: features/auth/, features/dashboard/, etc.",
            "- Keep components pure and testable",
            "- Use React Context for theme/auth state, not for data fetching",
            "- Implement proper component composition",
        ]
    
    else:
        lines += [
            "- Organize code by feature, not by layer",
            "- Keep business logic separate from presentation",
            "- Use environment variables for configuration",
            "- Implement proper error handling with try/catch",
        ]
    
    # Complex features (applies to all stacks)
    if complexity == Complexity.COMPLEX:
        features_str = " ".join(_normalize_features(features)).lower()
        
        if "dashboard" in features_str or "analytics" in features_str:
            lines += [
                "- Dashboard: implement data aggregation at API layer, not client",
                "- Use server-side pagination for all data tables",
                "- Charts: prefer lightweight recharts (Next.js) or chart.js",
            ]
        
        if "upload" in features_str or "file" in features_str or "image" in features_str:
            lines += [
                "- File uploads: validate type and size server-side before storage",
                "- Return presigned URLs for direct S3 uploads when possible",
                "- Store file metadata in database, not the file content",
            ]
        
        if "payment" in features_str or "stripe" in features_str:
            lines += [
                "- Payment: NEVER process payments client-side",
                "- Verify webhook signatures before processing payment events",
                "- Idempotency keys on all payment API calls",
                "- Store payment intent IDs in database for reconciliation",
            ]
        
        if "search" in features_str:
            lines += [
                "- Search: debounce client-side search input (300ms minimum)",
                "- Full-text search via database (LIKE/ILIKE for simple, pg_trgm for better)",
            ]
        
        if "chat" in features_str or "message" in features_str:
            lines += [
                "- Chat: implement WebSocket for real-time messaging",
                "- Store message history in database",
                "- Implement typing indicators and read receipts",
            ]
        
        if "api" in features_str:
            lines += [
                "- API: implement versioning (v1, v2)",
                "- Use rate limiting for public endpoints",
                "- Implement request validation middleware",
                "- Generate OpenAPI/Swagger documentation",
            ]
    
    return "\n".join(lines)

def build_system_prompt(
    app_name: str,
    description: str,
    features: list[str],
    auth_required: bool,
    database: str,
    stack: dict,
    include_agents_md: bool = True,
) -> str:
    """
    Build a complete system prompt with SkillOpt-optimized behaviors.
    
    Args:
        app_name: Name of the application
        description: Brief description of the project
        features: List of features to implement
        auth_required: Whether authentication is required
        database: Database type (e.g., 'postgresql', 'sqlite', 'mongodb')
        stack: Stack configuration dict with frontend/backend/extras
        include_agents_md: Whether to include AGENTS.md (default: True)
    
    Returns:
        Complete system prompt string for the AI agent
    """
    
    if not app_name or not description or not isinstance(stack, dict):
        raise ValueError("app_name, description, and stack are required")
    
    features_list = _normalize_features(features)
    stack_desc = get_stack_description(stack)
    flags = detect_stack_flags(stack)
    complexity = _classify_complexity(features_list, auth_required, database, stack)
    
    # ── Detect what the project actually needs ────────────────
    combined = (
        description + " " +
        " ".join(features_list) + " " +
        " ".join(_normalize_extras(stack.get("extras", [])))
    ).lower()
    
    has_alembic = "alembic" in combined or "migration" in combined
    has_websockets = any(x in combined for x in ("websocket", "realtime", "socket"))
    
    # ── Load AGENTS.md (AUTHORITATIVE GUIDE) ──────────────────
    agents_md = ""
    if include_agents_md:
        agents_md = load_agents_md()
    
    # ── Load optimized skill ────────────────────────────────────
    skill_content = _load_generation_skill()
    skill_version = _SKILL_VERSION
    
    # ── Build rules ─────────────────────────────────────────────
    rules: list[str] = []
    
    # ── 1. AGENTS.md (AUTHORITATIVE - ALWAYS FIRST) ──
    if agents_md:
        rules.append(f"""
# ═══════════════════════════════════════════════════════════
# AGENTS.MD — AUTHORITATIVE PROJECT STANDARDS (MUST FOLLOW)
# ═══════════════════════════════════════════════════════════
# This document overrides ALL other instructions.
# Every rule in AGENTS.md is MANDATORY.
# ═══════════════════════════════════════════════════════════

{agents_md}

# ═══════════════════════════════════════════════════════════
# END OF AGENTS.MD - ALL ABOVE RULES ARE MANDATORY
# ═══════════════════════════════════════════════════════════
""")
    
    # ── 2. IMMUTABLE STACK BLOCK ──
    rules.append(_get_immutable_stack_block(stack, auth_required))
    
    # ── 3. Critical files checklist ──
    rules.append(_get_critical_files_rules(stack, auth_required, database))
    
    # ── 4. Stack-specific rules ──
    stack_rules = _get_stack_specific_rules(stack, auth_required, database)
    if stack_rules:
        rules.append(stack_rules)
    
    # ── 5. Auth rules ──
    if auth_required:
        rules.append(_get_auth_rules(stack, complexity))
    
    # ── 6. API + DB init rules ──
    api_db_rules = _get_api_and_db_init_rules(stack, database)
    if api_db_rules:
        rules.append(api_db_rules)
    
    # ── 7. Architecture rules (for moderate/complex) ──
    arch_rules = _get_architecture_rules(features_list, stack, complexity)
    if arch_rules:
        rules.append(arch_rules)
    
    # ── 8. Universal rules ──
    rules.append(get_universal_rules())
    
    # ── 9. Anti-patterns ──
    rules.append(_get_anti_patterns(stack, complexity))
    
    # ── CDN-only project rule ──
    if flags.get("is_cdn_frontend") and not flags.get("is_node_frontend"):
        rules.append(
            "NO-NPM RULE: CDN-only project.\n"
            "- DO NOT generate package.json.\n"
            "- Load all JS via <script> tags in HTML."
        )

    # ── Assemble final prompt ──────────────────────────────────
    features_text = (
        "\n".join(f"  - {f}" for f in features_list)
        if features_list else "  - none specified"
    )
    
    extras = _normalize_extras(stack.get("extras", []))
    extras_text = (
        "\n".join(f"  - {e}" for e in extras)
        if extras else "  - none"
    )
    
    complexity_guidance = {
        Complexity.SIMPLE: "Focused, single-purpose app. Keep it clean and minimal.",
        Complexity.MODERATE: "Multi-feature app. Proper error handling, separation of concerns.",
        Complexity.COMPLEX: "Production-grade. Layered architecture, full error handling, no stubs.",
    }[complexity]
    
    rules_block = "\n\n".join(r for r in rules if r and r.strip())
    
    # ── EVOLVE: Add learned warnings ────────────────────────────
    try:
        from app.services.evolve_service import PromptAugmenter
        import asyncio
        loop = asyncio.get_event_loop()
        if loop.is_running():
            import nest_asyncio
            nest_asyncio.apply()
        learned = asyncio.run(PromptAugmenter.get_learned_additions(stack))
        if learned:
            rules_block += "\n\n# ═══════════════════════════════════════════════════════════\n"
            rules_block += "# LEARNED FROM PAST FAILURES (auto-generated)\n"
            rules_block += "# ═══════════════════════════════════════════════════════════\n"
            rules_block += learned
    except Exception:
        pass

    # ── Build the final prompt ──────────────────────────────────
    return f"""You are an expert {stack_desc} developer generating complete, production-ready code.

PROJECT: {app_name}
DESCRIPTION: {description}
COMPLEXITY: {complexity.value.upper()}

{complexity_guidance}

FEATURES:
{features_text}

EXTRAS:
{extras_text}

AUTH: {"YES" if auth_required else "NO"}
DATABASE: {database or "none"}

{rules_block}

# ═══════════════════════════════════════════════════════════
# OPTIMIZED SKILL (SkillOpt v{skill_version})
# ═══════════════════════════════════════════════════════════

{skill_content}

# ═══════════════════════════════════════════════════════════
# SECURITY REQUIREMENTS
# ═══════════════════════════════════════════════════════════

{_load_security_skill()}

CRITICAL — Config file layout:
- Settings MUST go in config/settings.py only. Never generate src/config/settings.py or src/core/config.py.
